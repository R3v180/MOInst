   504→
   505→function PriceHistoryDialog({
   506→  open,
   507→  onOpenChange,
   508→  articleId,
   509→}: {
   510→  open: boolean;
   511→  onOpenChange: (v: boolean) => void;
   512→  articleId: string;
   513→}) {
   514→  const { data, isLoading } = useQuery({
   515→    queryKey: ["price-history", articleId],
   516→    queryFn: () => fetch(`/api/articles/${articleId}/price-history`).then((r) => r.json()),
   517→    enabled: open && !!articleId,
   518→  });
   519→
   520→  const suppliers = data?.suppliers ?? [];
   521→  const rows = data?.rows ?? [];
   522→
   523→  const [off, setOff] = useState<Record<string, boolean>>({});
   524→  const isSelected = (sid: string) => !off[sid];
   525→
   526→  // Pivotar datos para recharts: cada priceDate → { date, [supplierId]: price }
   527→  const chartData = useMemo(() => {
   528→    const byDate = new Map<string, any>();
   529→    for (const r of rows) {
   530→      if (off[r.supplierId]) continue;
   531→      const key = new Date(r.priceDate).getTime();
   532→      const entry = byDate.get(key) ?? { ts: key, dateLabel: formatDate(r.priceDate) };
   533→      entry[r.supplierId] = r.price;
   534→      entry[`${r.supplierId}_name`] = r.supplierName;
   535→      byDate.set(key, entry);
   536→    }
   537→    return Array.from(byDate.values()).sort((a, b) => a.ts - b.ts);
   538→  }, [rows, off]);
   539→
   540→  const visibleSuppliers = suppliers.filter((s: any) => !off[s.id]);
   541→
   542→  return (
   543→    <Dialog open={open} onOpenChange={onOpenChange}>
   544→      <DialogContent className="max-w-4xl max-h-[90vh] overflow-y-auto scroll-thin">

   672→function AddSupplierDialog({
   673→  open,
   674→  onOpenChange,
   675→  articleInternalCode,
   676→  articleName,
   677→  existingSupplierIds,
   678→  onSubmit,
   679→  loading,
   680→}: {
   681→  open: boolean;
   682→  onOpenChange: (v: boolean) => void;
   683→  articleInternalCode: string;
   684→  articleName: string;
   685→  existingSupplierIds: string[];
   686→  onSubmit: (d: any) => void;
   687→  loading: boolean;
   688→}) {
   689→  const [q, setQ] = useState("");
   690→  const [selectedSupplier, setSelectedSupplier] = useState<string>("");
   691→  const [d, setD] = useState<any>({});
   692→  const set = (k: string, v: any) => setD((p) => ({ ...p, [k]: v }));
   693→
   694→  const { data: suppliersData, isLoading } = useQuery({
   695→    queryKey: ["suppliers", q, "for-article"],
   696→    queryFn: async () => {
   697→      const p = new URLSearchParams();
   698→      if (q) p.set("q", q);
   699→      p.set("pageSize", "50");
   700→      const r = await fetch(`/api/suppliers?${p.toString()}`);
   701→      return r.json();
   702→    },
   703→    enabled: open,
   704→  });
   705→
   706→  const reset = () => {
   707→    setQ("");
   708→    setSelectedSupplier("");
   709→    setD({});
   710→  };
   711→
   712→  const submit = () => {
   713→    if (!selectedSupplier) {
   714→      toast({ title: "Selecciona un proveedor", variant: "destructive" });
   715→      return;
   716→    }
   717→    if (d.price == null || Number(d.price) < 0) {
   718→      toast({ title: "Precio inválido", variant: "destructive" });
   719→      return;
   720→    }
   721→    onSubmit({
   722→      supplierId: selectedSupplier,
   723→      price: Number(d.price),
   724→      supplierRef: d.supplierRef ?? null,
   725→      deliveryDays: d.deliveryDays ? Number(d.deliveryDays) : null,
   726→      notes: d.notes ?? null,
   727→    });
   728→    reset();
   729→  };
   730→
   731→  const suppliers = (suppliersData?.items ?? []).filter(
   732→    (s: any) => !existingSupplierIds.includes(s.id)
   733→  );
   734→
   735→  return (
   736→    <Dialog
   737→      open={open}
   738→      onOpenChange={(v) => {
   739→        if (!v) reset();
   740→        onOpenChange(v);
   741→      }}
   742→    >
   743→      <DialogContent className="max-w-2xl max-h-[90vh] overflow-y-auto scroll-thin">
   744→        <DialogHeader>
   745→          <DialogTitle>Asociar proveedor y precio</DialogTitle>
   746→        </DialogHeader>
   747→        <div className="text-sm text-muted-foreground">
   748→          Artículo: <span className="font-medium text-foreground">{articleName}</span>{" "}
   749→          <span className="font-mono">({articleInternalCode})</span>
   750→        </div>
   751→
   752→        <div className="space-y-3">
   753→          <div>
   754→            <Label>Buscar proveedor</Label>
   755→            <Input
   756→              value={q}
   757→              onChange={(e) => setQ(e.target.value)}
   758→              placeholder="Nombre o contacto..."
   759→            />
   760→          </div>
   761→
   762→          {isLoading ? (
   763→            <div className="text-sm text-muted-foreground py-2">Cargando...</div>
   764→          ) : suppliers.length === 0 ? (
   765→            <p className="text-sm text-muted-foreground py-2">
   766→              No hay proveedores disponibles. Crea primero el proveedor.
   767→            </p>
   768→          ) : (
   769→            <div className="border border-border rounded-md max-h-48 overflow-y-auto scroll-thin">
   770→              {suppliers.map((s: any) => (
   771→                <label
   772→                  key={s.id}
   773→                  className="flex items-center gap-2 px-3 py-2 hover:bg-accent cursor-pointer border-b border-border last:border-b-0"
   774→                >
   775→                  <input
   776→                    type="radio"
   777→                    name="supplier"
   778→                    checked={selectedSupplier === s.id}
   779→                    onChange={() => setSelectedSupplier(s.id)}
   780→                  />
   781→                  <span className="font-medium text-sm">{s.name}</span>
   782→                  {s.contactName && (
   783→                    <span className="text-xs text-muted-foreground">· {s.contactName}</span>
   784→                  )}
   785→                </label>
   786→              ))}
   787→            </div>
   788→          )}
   789→
   790→          <div className="grid grid-cols-3 gap-3">
   791→            <div>
   792→              <Label>Precio (€) *</Label>
   793→              <Input
   794→                type="number"
   795→                step="0.01"
   796→                value={d.price ?? ""}
   797→                onChange={(e) => set("price", e.target.value)}
   798→                placeholder="0.00"
   799→              />
   800→            </div>
   801→            <div>
   802→              <Label>Días entrega</Label>
   803→              <Input
   804→                type="number"
   805→                value={d.deliveryDays ?? ""}
   806→                onChange={(e) => set("deliveryDays", e.target.value)}
   807→                placeholder="ej. 3"
   808→              />
   809→            </div>
   810→            <div>
   811→              <Label>Ref. proveedor</Label>
   812→              <Input
   813→                value={d.supplierRef ?? ""}
   814→                onChange={(e) => set("supplierRef", e.target.value)}
   815→              />
   816→            </div>
   817→          </div>
   818→          <div>
   819→            <Label>Notas</Label>
   820→            <Textarea
   821→              rows={2}
   822→              value={d.notes ?? ""}
   823→              onChange={(e) => set("notes", e.target.value)}
   824→              placeholder="ej. oferta válida hasta fin de mes"
   825→            />
   826→          </div>
   827→          <p className="text-xs text-muted-foreground">
   828→            Se añadirá una nueva fila al histórico (no se sobrescriben precios anteriores).
   829→          </p>
   830→        </div>
   831→
   832→        <DialogFooter>
   833→          <Button variant="outline" onClick={() => onOpenChange(false)}>Cancelar</Button>
   834→          <Button onClick={submit} disabled={loading}>
   835→            {loading ? <Loader2 className="w-4 h-4 mr-2 animate-spin" /> : <Plus className="w-4 h-4 mr-2" />}
   836→            Añadir precio
   837→          </Button>
   838→        </DialogFooter>
   839→      </DialogContent>
   840→    </Dialog>
   841→  );
   842→}
