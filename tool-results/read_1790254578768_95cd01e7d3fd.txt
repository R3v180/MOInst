     1→"use client";
     2→
     3→import { useEffect, useState } from "react";
     4→import { useQuery, useMutation, useQueryClient } from "@tanstack/react-query";
     5→import { useAppStore } from "@/store/app-store";
     6→import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
     7→import { Button } from "@/components/ui/button";
     8→import { Input } from "@/components/ui/input";
     9→import { Label } from "@/components/ui/label";
    10→import { Textarea } from "@/components/ui/textarea";
    11→import { Badge } from "@/components/ui/badge";
    12→import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from "@/components/ui/select";
    13→import { Dialog, DialogContent, DialogHeader, DialogTitle, DialogFooter } from "@/components/ui/dialog";
    14→import {
    15→  DropdownMenu,
    16→  DropdownMenuTrigger,
    17→  DropdownMenuContent,
    18→  DropdownMenuItem,
    19→  DropdownMenuSeparator,
    20→  DropdownMenuLabel,
    21→} from "@/components/ui/dropdown-menu";
    22→import { Table, TableHeader, TableBody, TableHead, TableRow, TableCell } from "@/components/ui/table";
    23→import { PageHeader } from "@/components/shared/page-header";
    24→import { StatusBadge } from "@/components/shared/status-badge";
    25→import { AttachmentUploader, AttachmentThumb } from "@/components/shared/attachment-uploader";
    26→import { formatCurrency, formatDate, fullAddress } from "@/lib/format";
    27→import {
    28→  ClipboardList, Loader2, ChevronDown, Wrench, FileText, FileInput,
    29→  Siren, Truck, Plus, Save, Package, Check,
    30→} from "lucide-react";
    31→import { toast } from "@/hooks/use-toast";
    32→import { Skeleton } from "@/components/ui/skeleton";
    33→
    34→type InstallItem =
    35→  | { mode: "new"; clientId: string; equipmentType: string; brand?: string | null; model?: string | null; serialNumber?: string | null; location?: string | null; warrantyEndDate?: string | null; notes?: string | null }
    36→  | { mode: "link"; installationId: string };
    37→
    38→export function SaleOrderDetailView() {
    39→  const { params, setView } = useAppStore();
    40→  const id = params.id;
    41→  const qc = useQueryClient();
    42→
    43→  const [showInstall, setShowInstall] = useState(false);
    44→  const [notesDraft, setNotesDraft] = useState<string | null>(null);
    45→
    46→  const { data: order, isLoading } = useQuery({
    47→    queryKey: ["sale-order", id],
    48→    queryFn: () => fetch(`/api/sale-orders/${id}`).then((r) => r.json()),
    49→    enabled: !!id,
    50→  });
    51→
    52→  // Lista de instalaciones del cliente (para vínculo)
    53→  const { data: clientInstallations } = useQuery({
    54→    queryKey: ["client-installations-for-order", order?.clientId],
    55→    queryFn: async () => {
    56→      // No existe /api/installations, pero podemos obtenerlas vía /api/clients/[id]
    57→      const r = await fetch(`/api/clients/${order.clientId}`);
    58→      if (!r.ok) return [];
    59→      const data = await r.json();
    60→      return (data.installations ?? []) as any[];
    61→    },
    62→    enabled: !!order?.clientId,
    63→  });
    64→
    65→  useEffect(() => {
    66→    if (order && notesDraft === null) {
    67→      // eslint-disable-next-line react-hooks/set-state-in-effect
    68→      setNotesDraft(order.notes ?? "");
    69→    }
    70→  }, [order, notesDraft]);
    71→
    72→  const updateMut = useMutation({
    73→    mutationFn: async (payload: any) => {
    74→      const r = await fetch(`/api/sale-orders/${id}`, {
    75→        method: "PUT",
    76→        headers: { "Content-Type": "application/json" },
    77→        body: JSON.stringify(payload),
    78→      });
    79→      if (!r.ok) {
    80→        const e = await r.json().catch(() => ({}));
    81→        throw new Error(e.error ?? "Error al actualizar");
    82→      }
    83→      return r.json();
    84→    },
    85→    onSuccess: (_d, vars) => {
    86→      toast({
    87→        title:
    88→          vars.status !== undefined ? "Estado actualizado"
    89→          : vars.paymentStatus !== undefined ? "Estado de pago actualizado"
    90→          : "Pedido actualizado",
    91→      });
    92→      qc.invalidateQueries({ queryKey: ["sale-order", id] });
    93→      qc.invalidateQueries({ queryKey: ["sale-orders"] });
    94→    },
    95→    onError: (e: any) => toast({ title: "Error", description: e.message, variant: "destructive" }),
    96→  });
    97→
    98→  const installMut = useMutation({
    99→    mutationFn: async (items: InstallItem[]) => {
   100→      const r = await fetch(`/api/sale-orders/${id}/install`, {
   101→        method: "POST",
   102→        headers: { "Content-Type": "application/json" },
   103→        body: JSON.stringify({ items }),
   104→      });
   105→      if (!r.ok) {
   106→        const e = await r.json().catch(() => ({}));
   107→        throw new Error(e.error ?? "Error al registrar instalación");
   108→      }
   109→      return r.json();
   110→    },
   111→    onSuccess: (data) => {
   112→      toast({
   113→        title: "Pedido instalado",
   114→        description: `${data.installations.length} instalación(es) registradas`,
   115→      });
   116→      setShowInstall(false);
   117→      qc.invalidateQueries({ queryKey: ["sale-order", id] });
   118→      qc.invalidateQueries({ queryKey: ["sale-orders"] });
   119→    },
   120→    onError: (e: any) => toast({ title: "Error", description: e.message, variant: "destructive" }),
   121→  });
   122→
   123→  if (isLoading || !order) {
   124→    return <div className="space-y-4"><Skeleton className="h-20" /><Skeleton className="h-96" /></div>;
   125→  }
   126→
   127→  const client = order.client;
   128→  const status = order.status;
   129→  const payment = order.paymentStatus;
   130→
   131→  const saveNotes = () => {
   132→    if (notesDraft !== null && notesDraft !== (order.notes ?? "")) {
   133→      updateMut.mutate({ notes: notesDraft || null });
   134→    }
   135→  };
   136→
   137→  return (
   138→    <div>
   139→      <PageHeader
   140→        title={order.number}
   141→        description={`Cliente: ${client?.name ?? "—"} · Emitido ${formatDate(order.issueDate)}`}
   142→        backTo="sale-orders"
   143→        actions={
   144→          <>
   145→            <DropdownMenu>
   146→              <DropdownMenuTrigger asChild>
   147→                <Button variant="outline" disabled={updateMut.isPending}>
   148→                  {updateMut.isPending ? <Loader2 className="w-4 h-4 mr-2 animate-spin" /> : <ChevronDown className="w-4 h-4 mr-2" />}
   149→                  Cambiar estado
   150→                </Button>
   151→              </DropdownMenuTrigger>
   152→              <DropdownMenuContent align="end">
   153→                <DropdownMenuLabel>Estado del pedido</DropdownMenuLabel>
   154→                <DropdownMenuSeparator />
   155→                {["PENDING", "IN_PROGRESS", "INSTALLED", "CLOSED"].map((s) => (
   156→                  <DropdownMenuItem
   157→                    key={s}
   158→                    onClick={() => updateMut.mutate({ status: s })}
   159→                    disabled={status === s}
   160→                    className="flex items-center justify-between"
   161→                  >
   162→                    <StatusBadge kind="saleOrder" value={s} />
   163→                    {status === s && <Check className="w-3.5 h-3.5" />}
   164→                  </DropdownMenuItem>
   165→                ))}
   166→              </DropdownMenuContent>
   167→            </DropdownMenu>
   168→
   169→            {status !== "INSTALLED" && status !== "CLOSED" && (
   170→              <Button onClick={() => setShowInstall(true)}>
   171→                <Wrench className="w-4 h-4 mr-2" /> Marcar instalado
   172→              </Button>
   173→            )}
   174→
   175→            <Button
   176→              variant={payment === "PAID" ? "default" : "outline"}
   177→              onClick={() => updateMut.mutate({ paymentStatus: payment === "PAID" ? "PENDING" : "PAID" })}
   178→              disabled={updateMut.isPending}
   179→            >
   180→              {payment === "PAID" ? (
   181→                <><Check className="w-4 h-4 mr-2" /> Cobrado</>
   182→              ) : (
   183→                <><Package className="w-4 h-4 mr-2" /> Marcar cobrado</>
   184→              )}
   185→            </Button>
   186→
   187→            <Button variant="outline" onClick={() => setView("incidents", { saleOrderId: id, clientId: order.clientId })}>
   188→              <Siren className="w-4 h-4 mr-2" /> Nueva incidencia
   189→            </Button>
   190→
   191→            <AttachmentUploader entityType="SALE_ORDER" entityId={id} label="Adjuntar albarán" onUploaded={() => qc.invalidateQueries({ queryKey: ["sale-order", id] })} />
   192→          </>
   193→        }
   194→      />
   195→
   196→      {/* Header card */}
   197→      <Card className="mb-4">
   198→        <CardContent className="p-4 sm:p-6 grid gap-4 sm:grid-cols-2 lg:grid-cols-4">
   199→          <div>
   200→            <div className="text-xs text-muted-foreground">Cliente</div>
   201→            <button
   202→              className="text-sm font-medium text-primary hover:underline truncate text-left"
   203→              onClick={() => setView("client-detail", { id: order.clientId })}
   204→            >
   205→              {client?.name ?? "—"}
   206→            </button>
   207→            <div className="text-xs text-muted-foreground mt-0.5">{fullAddress(client)}</div>
   208→          </div>
   209→          <div>
   210→            <div className="text-xs text-muted-foreground">Presupuesto de origen</div>
   211→            {order.sourceSaleQuote ? (
   212→              <button
   213→                className="text-sm font-medium text-primary hover:underline flex items-center gap-1"
   214→                onClick={() => setView("sale-quote-detail", { id: order.sourceSaleQuote.id })}
   215→              >
   216→                <FileText className="w-3.5 h-3.5" />
   217→                {order.sourceSaleQuote.number}
   218→              </button>
   219→            ) : (
   220→              <span className="text-sm text-muted-foreground">— Pedido directo —</span>
   221→            )}
   222→            {order.sourceSaleQuote && (
   223→              <div className="text-xs text-muted-foreground">Total presupuesto: {formatCurrency(order.sourceSaleQuote.total)}</div>
   224→            )}
   225→          </div>
   226→          <div>
   227→            <div className="text-xs text-muted-foreground">Emisión</div>
   228→            <div className="text-sm font-medium">{formatDate(order.issueDate)}</div>
   229→            {order.createdBy?.name && (
   230→              <div className="text-xs text-muted-foreground">Creado por {order.createdBy.name}</div>
   231→            )}
   232→          </div>
   233→          <div>
   234→            <div className="text-xs text-muted-foreground">Estado / Pago</div>
   235→            <div className="flex items-center gap-2 mt-0.5">
   236→              <StatusBadge kind="saleOrder" value={status} />
   237→              <StatusBadge kind="payment" value={payment} />
   238→            </div>
   239→          </div>
   240→        </CardContent>
   241→      </Card>
   242→
   243→      <div className="grid gap-4 lg:grid-cols-3">
   244→        {/* Columna principal */}
   245→        <div className="lg:col-span-2 space-y-4">
   246→          {/* Líneas */}
   247→          <Card>
   248→            <CardHeader className="pb-3">
   249→              <CardTitle className="text-base flex items-center gap-2">
   250→                <ClipboardList className="w-4 h-4 text-primary" /> Líneas
   251→              </CardTitle>
   252→            </CardHeader>
   253→            <CardContent className="p-0">
   254→              {(!order.lines || order.lines.length === 0) ? (
   255→                <div className="py-6 text-center text-sm text-muted-foreground">Sin líneas.</div>
   256→              ) : (
   257→                <div className="overflow-x-auto">
   258→                  <Table>
   259→                    <TableHeader>
   260→                      <TableRow>
   261→                        <TableHead>Descripción</TableHead>
   262→                        <TableHead className="w-20">Cant.</TableHead>
   263→                        <TableHead className="w-28">P. unit.</TableHead>
   264→                        <TableHead className="w-20">Desc.</TableHead>
   265→                        <TableHead className="w-32 text-right">Subtotal</TableHead>
   266→                      </TableRow>
   267→                    </TableHeader>
   268→                    <TableBody>
   269→                      {order.lines.map((l: any) => (
   270→                        <TableRow key={l.id} className={l.isLabor ? "bg-amber-50/40 dark:bg-amber-900/10" : ""}>
   271→                          <TableCell>
   272→                            <div className="text-sm font-medium">{l.description}</div>
   273→                            {l.isLabor && <Badge variant="outline" className="mt-0.5 text-amber-700 border-amber-300">Mano de obra</Badge>}
   274→                            {l.article && (
   275→                              <div className="text-xs text-muted-foreground">{l.article.internalCode}</div>
   276→                            )}
   277→                          </TableCell>
   278→                          <TableCell className="text-sm">{l.quantity}</TableCell>
   279→                          <TableCell className="text-sm">{formatCurrency(l.unitPrice)}</TableCell>
   280→                          <TableCell className="text-sm">{l.discount}%</TableCell>
   281→                          <TableCell className="text-right font-semibold">{formatCurrency(l.subtotal)}</TableCell>
   282→                        </TableRow>
   283→                      ))}
   284→                    </TableBody>
   285→                  </Table>
   286→                </div>
   287→              )}
   288→              {order.lines && order.lines.length > 0 && (
   289→                <div className="flex justify-end p-4 border-t border-border">
   290→                  <div className="text-right">
   291→                    <div className="text-xs text-muted-foreground">Total pedido</div>
   292→                    <div className="text-lg font-bold">
   293→                      {formatCurrency(order.lines.reduce((s: number, l: any) => s + l.subtotal, 0))}
   294→                    </div>
   295→                  </div>
   296→                </div>
   297→              )}
   298→            </CardContent>
   299→          </Card>
   300→
   301→          {/* Instalaciones generadas */}
   302→          <Card>
   303→            <CardHeader className="pb-3 flex-row items-center justify-between">
   304→              <CardTitle className="text-base flex items-center gap-2">
   305→                <Wrench className="w-4 h-4 text-primary" /> Instalaciones generadas
   306→              </CardTitle>
   307→              {status !== "INSTALLED" && status !== "CLOSED" && (
   308→                <Button size="sm" variant="outline" onClick={() => setShowInstall(true)}>
   309→                  <Plus className="w-4 h-4 mr-1" /> Vincular / crear
   310→                </Button>
   311→              )}
   312→            </CardHeader>
   313→            <CardContent className="p-0">
   314→              {order.installations.length === 0 ? (
   315→                <div className="py-6 text-center text-sm text-muted-foreground">
   316→                  No hay instalaciones vinculadas. Marca el pedido como instalado para crearlas.
   317→                </div>
   318→              ) : (
   319→                <ul className="divide-y divide-border">
   320→                  {order.installations.map((i: any) => (
   321→                    <li key={i.id}>
   322→                      <button
   323→                        onClick={() => setView("installation-detail", { id: i.id })}
   324→                        className="w-full text-left py-3 px-4 hover:bg-accent flex items-center gap-3"
   325→                      >
   326→                        <Wrench className="w-4 h-4 text-primary shrink-0" />
   327→                        <div className="flex-1 min-w-0">
   328→                          <div className="text-sm font-medium truncate">
   329→                            {[i.brand, i.model].filter(Boolean).join(" ") || i.equipmentType}
   330→                          </div>
   331→                          <div className="text-xs text-muted-foreground">
   332→                            {i.serialNumber ? `S/N ${i.serialNumber}` : ""} · {i.location ?? ""}
   333→                          </div>
   334→                        </div>
   335→                        <StatusBadge kind="installation" value={i.status} />
   336→                      </button>
   337→                    </li>
   338→                  ))}
   339→                </ul>
   340→              )}
   341→            </CardContent>
   342→          </Card>
   343→
   344→          {/* Albaranes de entrega */}
   345→          <Card>
   346→            <CardHeader className="pb-3 flex-row items-center justify-between">
   347→              <CardTitle className="text-base flex items-center gap-2">
   348→                <Truck className="w-4 h-4 text-primary" /> Albaranes de entrega
   349→              </CardTitle>
   350→              <AttachmentUploader entityType="SALE_ORDER" entityId={id} variant="compact" onUploaded={() => qc.invalidateQueries({ queryKey: ["sale-order", id] })} />
   351→            </CardHeader>
   352→            <CardContent>
   353→              {order.deliveryAlbaranes && order.deliveryAlbaranes.length > 0 ? (
   354→                <div className="space-y-3">
   355→                  {order.deliveryAlbaranes.map((a: any) => (
   356→                    <div key={a.id} className="border border-border rounded-md p-3">
   357→                      <div className="flex items-center justify-between mb-2">
   358→                        <div>
   359→                          <div className="text-sm font-medium">Albarán · {formatDate(a.date)}</div>
   360→                          <div className="text-xs text-muted-foreground">Subido por {a.uploadedBy?.name ?? "—"}</div>
   361→                        </div>
   362→                        <Badge variant="outline">{a.notes ?? "CLIENT_DELIVERY"}</Badge>
   363→                      </div>
   364→                      {a.attachments && a.attachments.length > 0 && (
   365→                        <div className="grid grid-cols-3 sm:grid-cols-4 md:grid-cols-6 gap-2">
   366→                          {a.attachments.map((f: any) => (
   367→                            <AttachmentThumb key={f.id} file={f} onDelete={async () => {
   368→                              await fetch(`/api/attachments/${f.id}`, { method: "DELETE" });
   369→                              qc.invalidateQueries({ queryKey: ["sale-order", id] });
   370→                            }} />
   371→                          ))}
   372→                        </div>
   373→                      )}
   374→                    </div>
   375→                  ))}
   376→                </div>
   377→              ) : order.attachments && order.attachments.length > 0 ? (
   378→                <div>
   379→                  <div className="text-xs text-muted-foreground mb-2">Adjuntos del pedido (no vinculados a albarán)</div>
   380→                  <div className="grid grid-cols-3 sm:grid-cols-4 md:grid-cols-6 gap-2">
   381→                    {order.attachments.map((f: any) => (
   382→                      <AttachmentThumb key={f.id} file={f} onDelete={async () => {
   383→                        await fetch(`/api/attachments/${f.id}`, { method: "DELETE" });
   384→                        qc.invalidateQueries({ queryKey: ["sale-order", id] });
   385→                      }} />
   386→                    ))}
   387→                  </div>
   388→                </div>
   389→              ) : (
   390→                <div className="py-6 text-center text-sm text-muted-foreground">
   391→                  Sin albaranes de entrega. Adjunta fotos o PDF del parte de entrega.
   392→                </div>
   393→              )}
   394→            </CardContent>
   395→          </Card>
   396→        </div>
   397→
   398→        {/* Columna lateral */}
   399→        <div className="space-y-4">
   400→          {/* Compras vinculadas */}
   401→          <Card>
   402→            <CardHeader className="pb-3">
   403→              <CardTitle className="text-base flex items-center gap-2">
   404→                <FileInput className="w-4 h-4 text-primary" /> Compras vinculadas
   405→              </CardTitle>
   406→            </CardHeader>
   407→            <CardContent className="space-y-3">
   408→              <div>
   409→                <div className="text-xs text-muted-foreground mb-1">Presupuestos de compra</div>
   410→                {order.purchaseQuotes.length === 0 ? (
   411→                  <p className="text-xs text-muted-foreground">Sin presupuestos de compra.</p>
   412→                ) : (
   413→                  <ul className="divide-y divide-border border border-border rounded-md">
   414→                    {order.purchaseQuotes.map((q: any) => (
   415→                      <li key={q.id}>
   416→                        <button
   417→                          onClick={() => setView("purchase-quote-detail", { id: q.id })}
   418→                          className="w-full text-left py-2 px-3 hover:bg-accent flex items-center gap-2"
   419→                        >
   420→                          <FileText className="w-3.5 h-3.5 text-primary shrink-0" />
   421→                          <div className="min-w-0 flex-1">
   422→                            <div className="text-sm font-medium truncate">{q.supplier?.name}</div>
   423→                            <div className="text-xs text-muted-foreground">{formatDate(q.issueDate)}</div>
   424→                          </div>
   425→                          <StatusBadge kind="purchaseQuote" value={q.status} />
   426→                        </button>
   427→                      </li>
   428→                    ))}
   429→                  </ul>
   430→                )}
   431→              </div>
   432→              <div>
   433→                <div className="text-xs text-muted-foreground mb-1">Pedidos de compra</div>
   434→                {order.purchaseOrders.length === 0 ? (
   435→                  <p className="text-xs text-muted-foreground">Sin pedidos de compra.</p>
   436→                ) : (
   437→                  <ul className="divide-y divide-border border border-border rounded-md">
   438→                    {order.purchaseOrders.map((p: any) => (
   439→                      <li key={p.id}>
   440→                        <button
   441→                          onClick={() => setView("purchase-order-detail", { id: p.id })}
   442→                          className="w-full text-left py-2 px-3 hover:bg-accent flex items-center gap-2"
   443→                        >
   444→                          <Package className="w-3.5 h-3.5 text-primary shrink-0" />
   445→                          <div className="min-w-0 flex-1">
   446→                            <div className="text-sm font-medium truncate">{p.number}</div>
   447→                            <div className="text-xs text-muted-foreground">{p.supplier?.name}</div>
   448→                          </div>
   449→                          <StatusBadge kind="purchaseOrder" value={p.status} />
   450→                        </button>
   451→                      </li>
   452→                    ))}
   453→                  </ul>
   454→                )}
   455→              </div>
   456→            </CardContent>
   457→          </Card>
   458→
   459→          {/* Incidencias */}
   460→          <Card>
   461→            <CardHeader className="pb-3 flex-row items-center justify-between">
   462→              <CardTitle className="text-base flex items-center gap-2">
   463→                <Siren className="w-4 h-4 text-destructive" /> Incidencias
   464→              </CardTitle>
   465→              <Button size="sm" variant="ghost" onClick={() => setView("incidents", { saleOrderId: id, clientId: order.clientId })}>
   466→                <Plus className="w-4 h-4 mr-1" /> Nueva
   467→              </Button>
   468→            </CardHeader>
   469→            <CardContent className="p-0">
   470→              {order.incidents.length === 0 ? (
   471→                <p className="text-sm text-muted-foreground py-4 text-center">Sin incidencias.</p>
   472→              ) : (
   473→                <ul className="divide-y divide-border">
   474→                  {order.incidents.map((i: any) => (
   475→                    <li key={i.id}>
   476→                      <button
   477→                        onClick={() => setView("incident-detail", { id: i.id })}
   478→                        className="w-full text-left py-2 px-4 hover:bg-accent flex items-center gap-2"
   479→                      >
   480→                        <Siren className="w-3.5 h-3.5 text-destructive shrink-0" />
   481→                        <div className="min-w-0 flex-1">
   482→                          <div className="text-sm font-medium truncate">{i.number}</div>
   483→                          <div className="text-xs text-muted-foreground truncate">{i.description}</div>
   484→                        </div>
   485→                        <StatusBadge kind="incident" value={i.status} />
   486→                      </button>
   487→                    </li>
   488→                  ))}
   489→                </ul>
   490→              )}
   491→            </CardContent>
   492→          </Card>
   493→
   494→          {/* Notas */}
   495→          <Card>
   496→            <CardHeader className="pb-3 flex-row items-center justify-between">
   497→              <CardTitle className="text-base">Notas</CardTitle>
   498→              <Button size="sm" variant="ghost" onClick={saveNotes} disabled={updateMut.isPending}>
   499→                {updateMut.isPending ? <Loader2 className="w-3.5 h-3.5 animate-spin" /> : <Save className="w-3.5 h-3.5" />}
   500→              </Button>
   501→            </CardHeader>
   502→            <CardContent>
   503→              <Textarea
   504→                value={notesDraft ?? ""}
   505→                onChange={(e) => setNotesDraft(e.target.value)}
   506→                onBlur={saveNotes}
   507→                rows={4}
   508→                placeholder="Notas internas del pedido..."
   509→              />
   510→            </CardContent>
   511→          </Card>
   512→        </div>
   513→      </div>
   514→
   515→      {/* Diálogo: marcar instalado */}
   516→      <InstallDialog
   517→        open={showInstall}
   518→        onOpenChange={setShowInstall}
   519→        loading={installMut.isPending}
   520→        clientId={order.clientId}
   521→        clientName={client?.name ?? ""}
   522→        availableInstallations={clientInstallations ?? []}
   523→        onSubmit={(items) => installMut.mutate(items)}
   524→      />
   525→    </div>
   526→  );
   527→}
   528→
   529→function InstallDialog({
   530→  open,
   531→  onOpenChange,
   532→  loading,
   533→  clientId,
   534→  clientName,
   535→  availableInstallations,
   536→  onSubmit,
   537→}: {
   538→  open: boolean;
   539→  onOpenChange: (v: boolean) => void;
   540→  loading: boolean;
   541→  clientId: string;
   542→  clientName: string;
   543→  availableInstallations: any[];
   544→  onSubmit: (items: InstallItem[]) => void;
   545→}) {
   546→  const [items, setItems] = useState<InstallItem[]>([
   547→    { mode: "new", clientId, equipmentType: "aire acondicionado", brand: "", model: "", serialNumber: "", location: "", warrantyEndDate: "", notes: "" },
   548→  ]);
   549→
   550→  useEffect(() => {
   551→    if (open) {
   552→      // eslint-disable-next-line react-hooks/set-state-in-effect
   553→      setItems([
   554→        { mode: "new", clientId, equipmentType: "aire acondicionado", brand: "", model: "", serialNumber: "", location: "", warrantyEndDate: "", notes: "" },
   555→      ]);
   556→    }
   557→  }, [open, clientId]);
   558→
   559→  const addNew = () => {
   560→    setItems((prev) => [
   561→      ...prev,
   562→      { mode: "new", clientId, equipmentType: "aire acondicionado", brand: "", model: "", serialNumber: "", location: "", warrantyEndDate: "", notes: "" },
   563→    ]);
   564→  };
   565→  const addLink = () => {
   566→    setItems((prev) => [...prev, { mode: "link", installationId: availableInstallations[0]?.id ?? "" }]);
   567→  };
   568→  const update = (idx: number, patch: any) => {
   569→    setItems((prev) => prev.map((it, i) => (i === idx ? { ...it, ...patch } as InstallItem : it)));
   570→  };
   571→  const remove = (idx: number) => {
   572→    setItems((prev) => prev.filter((_, i) => i !== idx));
   573→  };
   574→
   575→  const submit = () => {
   576→    // Validar
   577→    for (const it of items) {
   578→      if (it.mode === "new") {
   579→        if (!it.equipmentType || !it.clientId) {
   580→          toast({ title: "Faltan datos en una instalación nueva", variant: "destructive" });
   581→          return;
   582→        }
   583→      } else if (!it.installationId) {
   584→        toast({ title: "Selecciona una instalación a vincular", variant: "destructive" });
   585→        return;
   586→      }
   587→    }
   588→    onSubmit(items);
   589→  };
   590→
   591→  // Filtrar instalaciones ya vinculadas a este pedido
   592→  const available = availableInstallations.filter(
   593→    (i) => !items.some((it) => it.mode === "link" && it.installationId === i.id)
   594→  );
   595→
   596→  return (
   597→    <Dialog open={open} onOpenChange={onOpenChange}>
   598→      <DialogContent className="max-w-2xl max-h-[90vh] overflow-y-auto scroll-thin">
   599→        <DialogHeader>
   600→          <DialogTitle>Marcar pedido como instalado</DialogTitle>
   601→        </DialogHeader>
   602→        <p className="text-sm text-muted-foreground">
   603→          Cliente: <span className="font-medium text-foreground">{clientName}</span>. Registra los equipos instalados o vincula instalaciones ya existentes.
   604→        </p>
   605→
   606→        <div className="space-y-3">
   607→          {items.map((it, idx) => (
   608→            <div key={idx} className="border border-border rounded-md p-3 relative">
   609→              <div className="flex items-center justify-between mb-2">
   610→                <div className="text-xs uppercase text-muted-foreground font-semibold">
   611→                  {it.mode === "new" ? "Nueva instalación" : "Vincular existente"}
   612→                </div>
   613→                {items.length > 1 && (
   614→                  <Button size="sm" variant="ghost" onClick={() => remove(idx)}>
   615→                    Quitar
   616→                  </Button>
   617→                )}
   618→              </div>
   619→
   620→              <div className="flex items-center gap-2 mb-2">
   621→                <Label className="text-xs w-20">Modo</Label>
   622→                <Select
   623→                  value={it.mode}
   624→                  onValueChange={(v) => update(idx, v === "link" ? { mode: "link", installationId: available[0]?.id ?? "" } : { mode: "new", clientId, equipmentType: "aire acondicionado", brand: "", model: "", serialNumber: "", location: "", warrantyEndDate: "", notes: "" })}
   625→                >
   626→                  <SelectTrigger className="h-8"><SelectValue /></SelectTrigger>
   627→                  <SelectContent>
   628→                    <SelectItem value="new">Crear nueva</SelectItem>
   629→                    <SelectItem value="link">Vincular existente</SelectItem>
   630→                  </SelectContent>
   631→                </Select>
   632→              </div>
   633→
   634→              {it.mode === "new" ? (
   635→                <div className="grid grid-cols-2 gap-2">
   636→                  <div className="col-span-2">
   637→                    <Label className="text-xs">Tipo de equipo *</Label>
   638→                    <Select
   639→                      value={(it as any).equipmentType}
   640→                      onValueChange={(v) => update(idx, { equipmentType: v })}
   641→                    >
   642→                      <SelectTrigger className="h-8"><SelectValue /></SelectTrigger>
   643→                      <SelectContent>
   644→                        <SelectItem value="aire acondicionado">Aire acondicionado</SelectItem>
   645→                        <SelectItem value="caldera">Caldera</SelectItem>
   646→                        <SelectItem value="termo">Termo</SelectItem>
   647→                        <SelectItem value="otro">Otro</SelectItem>
   648→                      </SelectContent>
   649→                    </Select>
   650→                  </div>
   651→                  <div>
   652→                    <Label className="text-xs">Marca</Label>
   653→                    <Input value={(it as any).brand ?? ""} onChange={(e) => update(idx, { brand: e.target.value })} className="h-8" />
   654→                  </div>
   655→                  <div>
   656→                    <Label className="text-xs">Modelo</Label>
   657→                    <Input value={(it as any).model ?? ""} onChange={(e) => update(idx, { model: e.target.value })} className="h-8" />
   658→                  </div>
   659→                  <div>
   660→                    <Label className="text-xs">Nº de serie</Label>
   661→                    <Input value={(it as any).serialNumber ?? ""} onChange={(e) => update(idx, { serialNumber: e.target.value })} className="h-8" />
   662→                  </div>
   663→                  <div>
   664→                    <Label className="text-xs">Ubicación</Label>
   665→                    <Input value={(it as any).location ?? ""} onChange={(e) => update(idx, { location: e.target.value })} className="h-8" placeholder="Salón, cocina..." />
   666→                  </div>
   667→                  <div>
   668→                    <Label className="text-xs">Fin de garantía</Label>
   669→                    <Input type="date" value={(it as any).warrantyEndDate ?? ""} onChange={(e) => update(idx, { warrantyEndDate: e.target.value || null })} className="h-8" />
   670→                  </div>
   671→                  <div>
   672→                    <Label className="text-xs">Notas</Label>
   673→                    <Input value={(it as any).notes ?? ""} onChange={(e) => update(idx, { notes: e.target.value })} className="h-8" />
   674→                  </div>
   675→                </div>
   676→              ) : (
   677→                <div>
   678→                  <Label className="text-xs">Instalación a vincular *</Label>
   679→                  {available.length === 0 ? (
   680→                    <p className="text-xs text-muted-foreground mt-1">No hay instalaciones disponibles para vincular.</p>
   681→                  ) : (
   682→                    <Select
   683→                      value={(it as any).installationId}
   684→                      onValueChange={(v) => update(idx, { installationId: v })}
   685→                    >
   686→                      <SelectTrigger><SelectValue placeholder="Selecciona instalación" /></SelectTrigger>
   687→                      <SelectContent>
   688→                        {available.map((i: any) => (
   689→                          <SelectItem key={i.id} value={i.id}>
   690→                            {[i.brand, i.model, i.serialNumber].filter(Boolean).join(" ") || i.equipmentType}
   691→                          </SelectItem>
   692→                        ))}
   693→                      </SelectContent>
   694→                    </Select>
   695→                  )}
   696→                </div>
   697→              )}
   698→            </div>
   699→          ))}
   700→        </div>
   701→
   702→        <div className="flex gap-2">
   703→          <Button variant="outline" size="sm" onClick={addNew}><Plus className="w-4 h-4 mr-1" /> Otra nueva</Button>
   704→          <Button variant="outline" size="sm" onClick={addLink} disabled={available.length === 0}><Plus className="w-4 h-4 mr-1" /> Vincular existente</Button>
   705→        </div>
   706→
   707→        <DialogFooter>
   708→          <Button variant="outline" onClick={() => onOpenChange(false)}>Cancelar</Button>
   709→          <Button onClick={submit} disabled={loading || items.length === 0}>
   710→            {loading ? <Loader2 className="w-4 h-4 mr-2 animate-spin" /> : <Check className="w-4 h-4 mr-2" />}
   711→            Marcar instalado
   712→          </Button>
   713→        </DialogFooter>
   714→      </DialogContent>
   715→    </Dialog>
   716→  );
   717→}
