   190→   190→
   191→   191→            <AttachmentUploader entityType="SALE_ORDER" entityId={id} label="Adjuntar albarán" onUploaded={() => qc.invalidateQueries({ queryKey: ["sale-order", id] })} />

   230→   230→              <div className="text-xs text-muted-foreground">Creado por {order.createdBy.name}</div>

   233→   233→          <div>

   238→   238→            </div>

   243→   243→      <div className="grid gap-4 lg:grid-cols-3">

   250→   250→                <ClipboardList className="w-4 h-4 text-primary" /> Líneas

   253→   253→            <CardContent className="p-0">

   278→   278→                          <TableCell className="text-sm">{l.quantity}</TableCell>

   305→   305→                <Wrench className="w-4 h-4 text-primary" /> Instalaciones generadas

   529→   529→function InstallDialog({
   530→   530→  open,
   531→   531→  onOpenChange,
   532→   532→  loading,
   533→   533→  clientId,
   534→   534→  clientName,
   535→   535→  availableInstallations,
   536→   536→  onSubmit,
   537→   537→}: {
   538→   538→  open: boolean;
   539→   539→  onOpenChange: (v: boolean) => void;
   540→   540→  loading: boolean;
   541→   541→  clientId: string;
   542→   542→  clientName: string;
   543→   543→  availableInstallations: any[];
   544→   544→  onSubmit: (items: InstallItem[]) => void;
   545→   545→}) {
   546→   546→  const [items, setItems] = useState<InstallItem[]>([
   547→   547→    { mode: "new", clientId, equipmentType: "aire acondicionado", brand: "", model: "", serialNumber: "", location: "", warrantyEndDate: "", notes: "" },
   548→   548→  ]);
   549→   549→
   550→   550→  useEffect(() => {
   551→   551→    if (open) {
   552→   552→      // eslint-disable-next-line react-hooks/set-state-in-effect
   553→   553→      setItems([
   554→   554→        { mode: "new", clientId, equipmentType: "aire acondicionado", brand: "", model: "", serialNumber: "", location: "", warrantyEndDate: "", notes: "" },
   555→   555→      ]);
   556→   556→    }
   557→   557→  }, [open, clientId]);
   558→   558→
   559→   559→  const addNew = () => {
   560→   560→    setItems((prev) => [
   561→   561→      ...prev,
   562→   562→      { mode: "new", clientId, equipmentType: "aire acondicionado", brand: "", model: "", serialNumber: "", location: "", warrantyEndDate: "", notes: "" },
   563→   563→    ]);
   564→   564→  };

   567→   567→  };
   568→   568→  const update = (idx: number, patch: any) => {

   585→   585→        return;
   586→   586→      }

   588→   588→    onSubmit(items);
   589→   589→  };
   590→   590→
   591→   591→  // Filtrar instalaciones ya vinculadas a este pedido
   592→   592→  const available = availableInstallations.filter(
   593→   593→    (i) => !items.some((it) => it.mode === "link" && it.installationId === i.id)
   594→   594→  );
   595→   595→
   596→   596→  return (
   597→   597→    <Dialog open={open} onOpenChange={onOpenChange}>
   598→   598→      <DialogContent className="max-w-2xl max-h-[90vh] overflow-y-auto scroll-thin">

   608→   608→            <div key={idx} className="border border-border rounded-md p-3 relative">
   609→   609→              <div className="flex items-center justify-between mb-2">

   611→   611→                  {it.mode === "new" ? "Nueva instalación" : "Vincular existente"}

   614→   614→                  <Button size="sm" variant="ghost" onClick={() => remove(idx)}>
   615→   615→                    Quitar

   617→   617→                )}
   618→   618→              </div>
   619→   619→
   620→   620→              <div className="flex items-center gap-2 mb-2">
   621→   621→                <Label className="text-xs w-20">Modo</Label>

   623→   623→                  value={it.mode}
   624→   624→                  onValueChange={(v) => update(idx, v === "link" ? { mode: "link", installationId: available[0]?.id ?? "" } : { mode: "new", clientId, equipmentType: "aire acondicionado", brand: "", model: "", serialNumber: "", location: "", warrantyEndDate: "", notes: "" })}
   625→   625→                >

   627→   627→                  <SelectContent>
   628→   628→                    <SelectItem value="new">Crear nueva</SelectItem>

   637→   637→                    <Label className="text-xs">Tipo de equipo *</Label>
   638→   638→                    <Select
   639→   639→                      value={(it as any).equipmentType}
   640→   640→                      onValueChange={(v) => update(idx, { equipmentType: v })}
   641→   641→                    >
   642→   642→                      <SelectTrigger className="h-8"><SelectValue /></SelectTrigger>
   643→   643→                      <SelectContent>
   644→   644→                        <SelectItem value="aire acondicionado">Aire acondicionado</SelectItem>
